import { Injectable, Logger } from '@nestjs/common';
import { PrismaService } from '../prisma.service';
import { RedisCache } from '../cache.service';

@Injectable()
export class OrderProcessingService {
  private readonly logger = new Logger(OrderProcessingService.name);

  constructor(
    private db: PrismaService,
    private cache: RedisCache,
  ) {}

  async processCheckout(orderId: string, userId: string) {
    // Idempotency check with Redis (TTL: 10m)
    const lockKey = `order:lock:${orderId}`;
    const acquired = await this.cache.setNx(lockKey, 'LOCKED', 600);
    if (!acquired) throw new Error('Pedido ya en procesamiento.');

    // Atomic transaction with inventory decrement
    return await this.db.$transaction(async (tx) => {
      const order = await tx.orders.update({
        where: { id: orderId, status: 'PENDING' },
        data: { status: 'PAID', confirmedAt: new Date() },
      });
      this.logger.log(`Orden ${orderId} confirmada con éxito.`);
      return { success: true, order };
    });
  }
}

Principios de Trabajo

1. Código legible antes que astuto

Escribo código pensando en la persona que lo mantendrá en 6 meses. Nombres explícitos, funciones de responsabilidad única y tipado estricto.

2. Pruebas como red de seguridad

Las suites de pruebas automatizadas garantizan que refactorizar sea seguro y que la lógica crítica de negocio no falle ante casos borde.

3. Respeto por los recursos y la red

En Latinoamérica la conectividad móvil varía: optimizar el tamaño de los bundles, comprimir assets y minimizar llamadas a la red no es un lujo, es accesibilidad.

4. Colaboración y Git disciplinado

Commits semánticos y atómicos, ramas descriptivas y pull requests documentados para facilitar la revisión de código en equipo.
